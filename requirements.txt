# voxcpm2-tts-cached — pinned dependencies (see README "Dependency pinning")
#
# Policy:
#  - Pure-Python / app-level deps are hard-pinned (voxcpm, transformers,
#    runpod, soundfile) so future releases cannot silently change behavior.
#  - The torch family is NOT hard-pinned to a +cuNNN wheel: the base image
#    (runpod/pytorch:2.4.0-cuda12.4.1) ships torch 2.4.0; voxcpm requires
#    >=2.5.0 so pip upgrades inside the image, and the exact wheel pip picks
#    is index-time. We set floors only and let the base image's CUDA 12.4
#    constrain the wheel family.
#  - transformers==4.51.3 is sacred: voxcpm declares >=4.36.2 with no upper
#    bound, and newer transformers break LlamaTokenizerFast
#    (TypeError: Input must be a List[Union[str, AddedToken]]).

voxcpm==2.0.3
transformers==4.51.3
runpod==1.12.0
soundfile>=0.12.0

# torch family — floors only (see policy above)
